```tsx
'use client'
import { motion } from 'framer-motion'
import styles from './DifficultySelector.module.css'
import type { Difficulty } from '@/types'

interface Props {
  value: Difficulty
  onChange: (d: Difficulty) => void
}

const difficulties: { value: Difficulty; label: string; descriptor: string; testId: string }[] = [
  { value: 'easy', label: 'EASY', descriptor: '200 common words', testId: 'difficulty-easy' },
  { value: 'medium', label: 'MEDIUM', descriptor: '500 mixed words', testId: 'difficulty-medium' },
  { value: 'hard', label: 'HARD', descriptor: '1000+ rare words', testId: 'difficulty-hard' },
]

export function DifficultySelector({ value, onChange }: Props) {
  return (
    <div
      className={styles.container}
      data-testid="difficulty-selector"
      role="radiogroup"
      aria-label="Select difficulty"
    >
      {difficulties.map((d) => (
        <motion.button
          key={d.value}
          className={`${styles.card} ${value === d.value ? styles.selected : ''}`}
          data-testid={d.testId}
          onClick={() => onChange(d.value)}
          role="radio"
          aria-checked={value === d.value}
          whileHover={{ scale: 1.03 }}
          whileTap={{ scale: 0.97 }}
          animate={value === d.value ? { scale: [1, 1.05, 1] } : { scale: 1 }}
          transition={
            value === d.value
              ? { type: 'spring', stiffness: 400, damping: 20 }
              : { duration: 0.15 }
          }
        >
          <span className={styles.label}>{d.label}</span>
          <span className={styles.descriptor}>{d.descriptor}</span>
        </motion.button>
      ))}
    </div>
  )
}
```

import { render, screen, fireEvent } from '@testing-library/react'
import { DifficultySelector } from './index'

describe('DifficultySelector', () => {
  const mockOnChange = jest.fn()

  beforeEach(() => { mockOnChange.mockClear() })

  it('renders EASY, MEDIUM, HARD options', () => {
    render(<DifficultySelector value="easy" onChange={mockOnChange} />)
    expect(screen.getByText('EASY')).toBeInTheDocument()
    expect(screen.getByText('MEDIUM')).toBeInTheDocument()
    expect(screen.getByText('HARD')).toBeInTheDocument()
  })

  it('fires onChange with correct value on click', () => {
    render(<DifficultySelector value="easy" onChange={mockOnChange} />)
    fireEvent.click(screen.getByTestId('difficulty-medium'))
    expect(mockOnChange).toHaveBeenCalledWith('medium')
  })

  it('selected option has aria-checked="true"', () => {
    render(<DifficultySelector value="medium" onChange={mockOnChange} />)
    expect(screen.getByTestId('difficulty-medium')).toHaveAttribute('aria-checked', 'true')
  })

  it('unselected options have aria-checked="false"', () => {
    render(<DifficultySelector value="easy" onChange={mockOnChange} />)
    expect(screen.getByTestId('difficulty-medium')).toHaveAttribute('aria-checked', 'false')
    expect(screen.getByTestId('difficulty-hard')).toHaveAttribute('aria-checked', 'false')
  })

  it('has data-testid="difficulty-selector" on container', () => {
    render(<DifficultySelector value="easy" onChange={mockOnChange} />)
    expect(screen.getByTestId('difficulty-selector')).toBeInTheDocument()
  })

  it('has correct testids on all cards', () => {
    render(<DifficultySelector value="easy" onChange={mockOnChange} />)
    expect(screen.getByTestId('difficulty-easy')).toBeInTheDocument()
    expect(screen.getByTestId('difficulty-medium')).toBeInTheDocument()
    expect(screen.getByTestId('difficulty-hard')).toBeInTheDocument()
  })

  it('renders no stat bar or progress elements inside cards', () => {
    render(<DifficultySelector value="easy" onChange={mockOnChange} />)
    expect(screen.queryByRole('progressbar')).not.toBeInTheDocument()
  })
})
