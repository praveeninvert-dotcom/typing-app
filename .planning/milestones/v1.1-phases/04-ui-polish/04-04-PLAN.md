'use client'
import { useEffect, useRef, useState } from 'react'
import { motion } from 'framer-motion'
import styles from './ResultOverlay.module.css'

// Preserve existing prop interface — adjust names to match actual types
interface Props {
  wpm: number
  accuracy: number
  correctChars: number
  wrongChars: number
  onRetry: () => void
  onHome: () => void
}

// Count-up animation hook
function useCountUp(target: number, duration: number, delay = 0): number {
  const [value, setValue] = useState(0)
  useEffect(() => {
    let frame: number
    const timeout = setTimeout(() => {
      let startTime: number | null = null
      const animate = (ts: number) => {
        if (!startTime) startTime = ts
        const elapsed = ts - startTime
        const progress = Math.min(elapsed / duration, 1)
        const eased = 1 - Math.pow(1 - progress, 2) // easeOut
        setValue(Math.round(target * eased))
        if (progress < 1) frame = requestAnimationFrame(animate)
      }
      frame = requestAnimationFrame(animate)
    }, delay)
    return () => { clearTimeout(timeout); cancelAnimationFrame(frame) }
  }, [target, duration, delay])
  return value
}

const backdropVariants = {
  hidden: { opacity: 0 },
  visible: { opacity: 1, transition: { duration: 0.2 } },
}
const modalVariants = {
  hidden: { opacity: 0, scale: 0.8 },
  visible: { opacity: 1, scale: 1, transition: { type: 'spring' as const, stiffness: 280, damping: 22 } },
}
const fadeIn = (delay: number) => ({
  hidden: { opacity: 0 },
  visible: { opacity: 1, transition: { delay, duration: 0.3 } },
})

export function ResultOverlay({ wpm, accuracy, correctChars, wrongChars, onRetry, onHome }: Props) {
  const retryRef = useRef<HTMLButtonElement>(null)
  const displayWpm = useCountUp(wpm, 800)
  const displayAcc = useCountUp(accuracy, 600, 100)

  // Focus retry button on mount (R-074)
  useEffect(() => { retryRef.current?.focus() }, [])

  // Enter key → onRetry
  useEffect(() => {
    const handler = (e: KeyboardEvent) => { if (e.key === 'Enter') onRetry() }
    window.addEventListener('keydown', handler)
    return () => window.removeEventListener('keydown', handler)
  }, [onRetry])

  // Character-by-character header reveal (preserve existing implementation or use this)
  const header = 'TEST COMPLETE'
  const [revealedChars, setRevealedChars] = useState(0)
  useEffect(() => {
    setRevealedChars(0)
    let i = 0
    const interval = setInterval(() => {
      i++
      setRevealedChars(i)
      if (i >= header.length) clearInterval(interval)
    }, 30)
    return () => clearInterval(interval)
  }, [])

  return (
    <motion.div
      className={styles.backdrop}
      data-testid="result-backdrop"
      variants={backdropVariants}
      initial="hidden"
      animate="visible"
    >
      <motion.div
        className={styles.modal}
        data-testid="result-modal"
        variants={modalVariants}
        initial="hidden"
        animate="visible"
      >
        <h2 className={styles.header} aria-label="Test Complete">
          {header.split('').map((ch, i) => (
            <span key={i} style={{ opacity: i < revealedChars ? 1 : 0 }}>
              {ch === ' ' ? '\u00A0' : ch}
            </span>
          ))}
        </h2>

        <motion.div className={styles.statsRow} variants={fadeIn(0.4)} initial="hidden" animate="visible">
          <div className={styles.statBox}>
            <span className={styles.wpmValue} data-testid="result-wpm">{displayWpm}</span>
            <span className={styles.statLabel}>WPM</span>
          </div>
          <motion.div className={`${styles.statBox} ${styles.accBox}`} variants={fadeIn(0.48)} initial="hidden" animate="visible">
            <span className={styles.accValue} data-testid="result-acc">{displayAcc}%</span>
            <span className={`${styles.statLabel} ${styles.accLabel}`}>ACCURACY</span>
          </motion.div>
        </motion.div>

        <motion.p
          className={styles.charBreakdown}
          data-testid="result-chars"
          variants={fadeIn(0.6)}
          initial="hidden"
          animate="visible"
        >
          <span className={styles.correct}>{correctChars}</span>
          <span className={styles.separator}> correct · </span>
          <span className={styles.wrong}>{wrongChars}</span>
          <span className={styles.separator}> wrong</span>
        </motion.p>

        <motion.hr className={styles.divider} variants={fadeIn(0.7)} initial="hidden" animate="visible" />

        <motion.p className={styles.playAgain} variants={fadeIn(0.8)} initial="hidden" animate="visible">
          PLAY AGAIN?
        </motion.p>

        <motion.div className={styles.buttonsRow} variants={fadeIn(0.9)} initial="hidden" animate="visible">
          <button
            ref={retryRef}
            className={styles.pixelButton}
            data-testid="retry-button"
            onClick={onRetry}
            type="button"
          >
            RETRY
          </button>
          <button
            className={styles.pixelButton}
            data-testid="home-button"
            onClick={onHome}
            type="button"
          >
            HOME
          </button>
        </motion.div>
      </motion.div>
    </motion.div>
  )
}
