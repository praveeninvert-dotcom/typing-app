Props interface:
```typescript
interface DifficultySelectorProps {
  selected: Difficulty | null
  onChange: (difficulty: Difficulty) => void
  disabled?: boolean
}
```

The three options:
```typescript
const DIFFICULTY_OPTIONS = [
  { value: 'easy' as const, label: 'EASY', descriptor: '200 common words' },
  { value: 'medium' as const, label: 'MEDIUM', descriptor: '500 words + punctuation' },
  { value: 'hard' as const, label: 'HARD', descriptor: '1000 words + numbers + symbols' },
]
```

ARIA radiogroup pattern (per R-075, CONTEXT.md locked decision, RESEARCH.md Pattern 7):
- Container: `role="radiogroup"` `aria-label="Select difficulty"`
- Each option: `role="radio"` `aria-checked={selected === diff.value}`
- Roving tabindex: `tabIndex={selected === diff.value || (selected === null && index === 0) ? 0 : -1}`
- Arrow key navigation: onKeyDown on the container handles ArrowDown/ArrowRight (next) and ArrowUp/ArrowLeft (prev), cycling circularly
- Enter or Space on an option: selects it and calls onChange

Full JSX structure:
```tsx
'use client'
import { useCallback } from 'react'
import type { Difficulty } from '@/types'
import styles from './DifficultySelector.module.css'

export function DifficultySelector({ selected, onChange, disabled }: DifficultySelectorProps) {
  const handleContainerKeyDown = useCallback((e: React.KeyboardEvent) => {
    const currentIndex = DIFFICULTY_OPTIONS.findIndex(d => d.value === selected) ?? 0
    if (e.key === 'ArrowDown' || e.key === 'ArrowRight') {
      e.preventDefault()
      const next = (currentIndex + 1) % DIFFICULTY_OPTIONS.length
      onChange(DIFFICULTY_OPTIONS[next].value)
    } else if (e.key === 'ArrowUp' || e.key === 'ArrowLeft') {
      e.preventDefault()
      const prev = (currentIndex - 1 + DIFFICULTY_OPTIONS.length) % DIFFICULTY_OPTIONS.length
      onChange(DIFFICULTY_OPTIONS[prev].value)
    }
  }, [selected, onChange])

  return (
    <div
      role="radiogroup"
      aria-label="Select difficulty"
      className={styles.group}
      onKeyDown={handleContainerKeyDown}
    >
      {DIFFICULTY_OPTIONS.map((diff, index) => (
        <div
          key={diff.value}
          role="radio"
          aria-checked={selected === diff.value}
          tabIndex={selected === diff.value || (selected === null && index === 0) ? 0 : -1}
          className={`${styles.option} ${selected === diff.value ? styles.selected : ''} ${disabled ? styles.disabled : ''}`}
          onClick={() => !disabled && onChange(diff.value)}
          onKeyDown={(e) => {
            if (e.key === 'Enter' || e.key === ' ') {
              e.preventDefault()
              if (!disabled) onChange(diff.value)
            }
          }}
        >
          <span className={styles.label}>{diff.label}</span>
          <span className={styles.descriptor}>{diff.descriptor}</span>
        </div>
      ))}
    </div>
  )
}
```

- `.valueAmber`: color var(--color-amber)
- `.valueBlue`: color var(--color-blue)
- `.valueWhite`: color #f0f0f0

Use `style={{ opacity: started ? 1 : 0, pointerEvents: started ? 'auto' : 'none' }}` on the root div. The transition animation (fade in on keypress) is added in Phase 2 — for Phase 1 it just snaps.

```tsx
'use client'
import styles from './StatsBar.module.css'

export function StatsBar({ wpm, timeLeft, accuracy, started }: StatsBarProps) {
  return (
    <div
      className={styles.bar}
      style={{ opacity: started ? 1 : 0, pointerEvents: started ? 'auto' : 'none' }}
      aria-hidden={!started}
    >
      <div className={styles.slot}>
        <span className={styles.label}>WPM</span>
        <span className={`${styles.value} ${styles.valueAmber}`}>{wpm}</span>
      </div>
      <div className={styles.slot}>
        <span className={styles.label}>TIME</span>
        <span className={`${styles.value} ${styles.valueBlue}`}>{timeLeft}</span>
      </div>
      <div className={styles.slot}>
        <span className={styles.label}>ACC</span>
        <span className={`${styles.value} ${styles.valueWhite}`}>
          {accuracy !== null ? `${accuracy}%` : '—'}
        </span>
      </div>
    </div>
  )
}
```

This is the performance-critical component. Must use React.memo on both WordSpan and CharSpan per RESEARCH.md Pattern 4.

Props:
```typescript
interface TextDisplayProps {
  words: Word[]
  currentWordIndex: number
  currentCharIndex: number
}
```

CharSpan (inner, memoized):
- Renders a single `<span>` with class based on char.state
- Adds cursor class when `isCursor` is true
- The cursor class adds `border-left: 2px solid var(--color-amber)` with blink animation

WordSpan (memoized):
- Renders a `<span>` containing all CharSpans for one word
- Active word: amber bottom-border class
- Incorrect completed word: dim red background tint class
- Followed by a space span

Key implementation notes from RESEARCH.md:
- Wrap CharSpan in `React.memo` — prevents re-render if props unchanged
- Wrap WordSpan in `React.memo` — only the active word and the word that just changed re-render
- `isCursor` computed at WordSpan level: `isActive && charIndex === currentCharIndex`
- Space at currentCharIndex === word.chars.length shows cursor after last char

Full component structure:
```tsx
'use client'
import { memo } from 'react'
import type { Word, Char } from '@/types'
import styles from './TextDisplay.module.css'

interface CharSpanProps {
  char: Char
  isCursor: boolean
}
const CharSpan = memo(function CharSpan({ char, isCursor }: CharSpanProps) {
  return (
    <span className={`${styles.char} ${styles[char.state]} ${isCursor ? styles.cursor : ''}`}>
      {char.expected}
    </span>
  )
})
CharSpan.displayName = 'CharSpan'

interface WordSpanProps {
  word: Word
  isActive: boolean
  currentCharIndex: number
}
const WordSpan = memo(function WordSpan({ word, isActive, currentCharIndex }: WordSpanProps) {
  return (
    <>
      <span className={`${styles.word} ${isActive ? styles.active : ''} ${word.state === 'incorrect' ? styles.wrongWord : ''}`}>
        {word.chars.map((char, charIndex) => (
          <CharSpan
            key={charIndex}
            char={char}
            isCursor={isActive && charIndex === currentCharIndex}
          />
        ))}
        {/* Cursor after last char: when currentCharIndex === word.chars.length */}
        {isActive && currentCharIndex === word.chars.length && (
          <span className={`${styles.char} ${styles.cursor}`}>{' '}</span>
        )}
      </span>
      <span className={styles.space}>{' '}</span>
    </>
  )
})
WordSpan.displayName = 'WordSpan'

export function TextDisplay({ words, currentWordIndex, currentCharIndex }: TextDisplayProps) {
  return (
    <div className={styles.container}>
      {words.map((word, wordIndex) => (
        <WordSpan
          key={wordIndex}
          word={word}
          isActive={wordIndex === currentWordIndex}
          currentCharIndex={wordIndex === currentWordIndex ? currentCharIndex : 0}
        />
      ))}
    </div>
  )
}
```
